    var center=track.scrollLeft+(track.clientWidth/2);
    var best=0,bestDist=Infinity;
    slides.forEach(function(slide,i){
      var slideCenter=slide.offsetLeft+(slide.offsetWidth/2);
      var dist=Math.abs(center-slideCenter);
      if(dist<bestDist){bestDist=dist;best=i;}
    });
    return best;
  }

  function updateDots(){
    var idx=nearestIndex();
    dots.forEach(function(dot,i){
      dot.classList.toggle('is-active',i===idx);
    });
  }

  function targetLeft(index){
    var slide=slides[index];
    var left=slide.offsetLeft-(track.clientWidth-slide.offsetWidth)/2;
    var maxLeft=Math.max(0,track.scrollWidth-track.clientWidth);
    return Math.max(0,Math.min(maxLeft,left));
  }

  function goTo(index){
    var target=((index%slides.length)+slides.length)%slides.length;
    track.scrollTo({
      left:targetLeft(target),
      behavior:'smooth'
    });
  }

  function go(delta){
    var idx=nearestIndex();
    var target=idx+delta;
    if(target>=slides.length)target=0;
    if(target<0)target=slides.length-1;
    goTo(target);
  }

  function stopAuto(){
    if(autoTimer){
      window.clearInterval(autoTimer);
      autoTimer=null;
    }
  }

  function startAuto(){
    stopAuto();
    if(!sectionVisible || userInteracting)return;
    if(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches)return;

    autoTimer=window.setInterval(function(){
      if(sectionVisible && !userInteracting){
        go(1);
      }
    },AUTO_DELAY);
  }

  function restartAuto(){
    stopAuto();
    if(sectionVisible && !userInteracting){
      window.setTimeout(startAuto,700);
    }
  }

  prev.addEventListener('click',function(){
    go(-1);
    restartAuto();
  });

  next.addEventListener('click',function(){
    go(1);
    restartAuto();
  });

  track.addEventListener('scroll',function(){
    window.clearTimeout(track._usoTimer);
    track._usoTimer=window.setTimeout(updateDots,80);
  },{passive:true});

  track.addEventListener('pointerdown',function(){
    userInteracting=true;
    stopAuto();
  });

  track.addEventListener('pointerup',function(){
    userInteracting=false;
    restartAuto();
  });

  track.addEventListener('pointercancel',function(){
    userInteracting=false;
    restartAuto();
  });

  track.addEventListener('touchstart',function(){
    userInteracting=true;
    stopAuto();
  },{passive:true});

  track.addEventListener('touchend',function(){
    userInteracting=false;
    restartAuto();
  },{passive:true});

  if('IntersectionObserver' in window){
    var observer=new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        sectionVisible=entry.isIntersecting && entry.intersectionRatio>=0.35;
        if(sectionVisible){
          startAuto();
        }else{
          stopAuto();
        }
      });
    },{
      threshold:[0,0.35,0.6]
    });
    observer.observe(section);
  }else{
    sectionVisible=true;
    startAuto();
  }

  document.addEventListener('visibilitychange',function(){
    if(document.hidden){
      stopAuto();
    }else if(sectionVisible){
      startAuto();
    }
  });

  updateDots();
})();



  (function () {
    var root = document.documentElement;
    if (!root || root.getAttribute("data-mb-utm-ready") === "true") return;
    root.setAttribute("data-mb-utm-ready", "true");

    /* Checkout de la plataforma. Relativo a prop\u00f3sito: se resuelve solo
       contra el dominio de la tienda. */
    var checkoutUrl = "/checkout";

    var campaignParamNames = ["utm_source", "utm_campaign", "utm_medium", "utm_content", "utm_term", "utm_id"];
    var campaignStorageKey = "bw3d_biblical_world_utm";

    function getCampaignParams() {
      var currentParams = new URLSearchParams(window.location.search);
      var campaignParams = {};
      var hasIncomingUtm = campaignParamNames.some(function (paramName) {
        return currentParams.has(paramName);
      });
      if (hasIncomingUtm) {
        campaignParamNames.forEach(function (paramName) {
          campaignParams[paramName] = currentParams.has(paramName) ? currentParams.get(paramName) : "";
        });
        try {
          window.sessionStorage.setItem(campaignStorageKey, JSON.stringify(campaignParams));
        } catch (error) {
          /* La navegaci\u00f3n contin\u00faa aunque el navegador bloquee sessionStorage. */
        }
      } else {
        try {
          campaignParams = JSON.parse(window.sessionStorage.getItem(campaignStorageKey) || "{}");
        } catch (error) {
          campaignParams = {};
        }
      }
      if (!campaignParams.utm_source) campaignParams.utm_source = "organic";
      campaignParamNames.forEach(function (paramName) {
        if (typeof campaignParams[paramName] !== "string") campaignParams[paramName] = "";
      });
      return campaignParams;
    }

    function decorateCheckoutLink(buyLink) {
      var destination = new URL(checkoutUrl, window.location.origin);
      var campaignParams = getCampaignParams();
      campaignParamNames.forEach(function (paramName) {
        destination.searchParams.set(paramName, campaignParams[paramName]);
      });
      buyLink.href = destination.toString();
    }

    document.querySelectorAll("[data-mb-buy]").forEach(function (buyLink) {
      decorateCheckoutLink(buyLink);
      buyLink.addEventListener("click", function () {
        decorateCheckoutLink(buyLink);
        if (Array.isArray(window.dataLayer)) {
          window.dataLayer.push({
            event: "bw3d_purchase_start",
            product: "biblical_world_3d_6_bonuses"
          });
        }
      });
    });
  }());
